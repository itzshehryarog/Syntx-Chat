local function Handler(Data)
    if not Data then
        return ""
    end

    local Entry = Data:gsub("<[^>]*>", "")
        :gsub("[<>]", "")
        :gsub("[\204-\205][\128-\191]", "")
        :gsub("%s+", " ")
    return Entry:match("^%s*(.-)%s*$") or Entry
end

local function Action(Input)
    if Input:find("<[^>]*>") then
        return true, "HTML tags not allowed"
    end

    if Input:find("[<>]") then
        return true, "Angle brackets not allowed"
    end

    if Input:find("[\204-\205][\128-\191]") then
        return true, "Zalgo text not allowed"
    end

    local Ref = Input:gsub("[^%a]", "")

    if #Ref > 0 then
        if Ref:match("(.)%1{15,}") then
            return true, "Too many repeated letters"
        end
    end

    if #Ref > 10 then
        local Node = select(2, Ref:gsub("%u", ""))
        if Node / #Ref > 0.8 and Node > 15 then
            return true, "Too many capital letters"
        end
    end

    return false, Input
end
